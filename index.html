Kode Backend Google Apps Script (Drive & Spreadsheet)

Otomatis membuat 2 Folder ("Folder Surat Masuk" & "Folder Surat Keluar") dan Sheet Database

Cara Pemasangan Singkat:

  1. Buka Google Spreadsheet baru → klik menu Ekstensi > Apps Script.
  2. Hapus kode lama, lalu Salin & Tempel seluruh kode di bawah ini.
  3. Klik Terapkan (Deploy) > Deployment Baru > Aplikasi Web (Akses: Siapa saja / Anyone).
  4. Salin URL Web App lalu tempel di menu Pengaturan aplikasi SIMDITAS ini.
// ============================================================================
// BACKEND GOOGLE APPS SCRIPT - SIMDITAS SMP NEGERI 3 BLANAKAN
// Otomatis membuat Folder Surat Masuk, Folder Surat Keluar, dan Sheet Database
// ============================================================================

const ROOT_FOLDER_NAME = "ARSIP_SURAT_SMPN3_BLANAKAN";
const FOLDER_MASUK_NAME = "Folder Surat Masuk";
const FOLDER_KELUAR_NAME = "Folder Surat Keluar";

function getOrCreateFolder(parent, name) {
  const iter = parent.getFoldersByName(name);
  return iter.hasNext() ? iter.next() : parent.createFolder(name);
}

function setupEnvironment() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  const root = getOrCreateFolder(DriveApp.getRootFolder(), ROOT_FOLDER_NAME);
  const folderMasuk = getOrCreateFolder(root, FOLDER_MASUK_NAME);
  const folderKeluar = getOrCreateFolder(root, FOLDER_KELUAR_NAME);

  const sheetsConfig = {
    "SuratMasuk": ["ID", "TanggalDiterima", "AsalSurat", "NomorSurat", "Indeks", "Perihal", "TanggalSurat", "JenisSurat", "Ditujukan", "Keterangan", "NamaFile", "LinkDrive"],
    "SuratKeluar": ["ID", "TanggalPembukuan", "NomorSurat", "Indeks", "Perihal", "TanggalSurat", "JenisSurat", "Ditujukan", "Keterangan", "NamaFile", "LinkDrive"],
    "Disposisi": ["NomorDisposisi", "NomorAgenda", "NomorSurat", "Perihal", "Pengirim", "Ditujukan", "Instruksi", "TindakLanjut", "TanggalDisposisi", "NamaKepalaSekolah", "NIP", "PangkatGolongan"]
  };

  Object.keys(sheetsConfig).forEach(sheetName => {
    let sheet = ss.getSheetByName(sheetName);
    if (!sheet) {
      sheet = ss.insertSheet(sheetName);
      sheet.appendRow(sheetsConfig[sheetName]);
      sheet.getRange(1, 1, 1, sheetsConfig[sheetName].length).setFontWeight("bold");
    }
  });

  return { ss, folderMasuk, folderKeluar };
}

function getSheetRecords(ss, sheetName) {
  const sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  const lastRow = sheet.getLastRow();
  const lastCol = sheet.getLastColumn();
  if (lastRow <= 1 || lastCol === 0) return [];
  const values = sheet.getRange(1, 1, lastRow, lastCol).getValues();
  const headers = values[0];
  const tz = Session.getScriptTimeZone();
  const records = [];
  for (let r = 1; r < values.length; r++) {
    const row = values[r];
    let hasValue = false;
    const obj = {};
    for (let c = 0; c < headers.length; c++) {
      let val = row[c];
      if (val instanceof Date) {
        val = Utilities.formatDate(val, tz, "yyyy-MM-dd");
      } else {
        val = val !== undefined && val !== null ? String(val) : "";
      }
      if (!hasValue && val.trim() !== "") hasValue = true;
      obj[headers[c]] = val;
    }
    if (hasValue) records.push(obj);
  }
  return records;
}

function doGet(e) {
  try {
    // Langsung baca Spreadsheet tanpa memanggil DriveApp agar respons 3x-5x lebih cepat
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const action = (e && e.parameter && e.parameter.action) ? e.parameter.action : "GET_ALL";
    if (action === "GET_MASUK") {
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        data: getSheetRecords(ss, "SuratMasuk")
      })).setMimeType(ContentService.MimeType.JSON);
    }
    if (action === "GET_KELUAR") {
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        data: getSheetRecords(ss, "SuratKeluar")
      })).setMimeType(ContentService.MimeType.JSON);
    }
    if (action === "GET_DISPOSISI") {
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        data: getSheetRecords(ss, "Disposisi")
      })).setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({
      status: "success",
      suratMasuk: getSheetRecords(ss, "SuratMasuk"),
      suratKeluar: getSheetRecords(ss, "SuratKeluar"),
      disposisi: getSheetRecords(ss, "Disposisi")
    })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    const ss = SpreadsheetApp.getActiveSpreadsheet();

    // Jalur cepat baca data tanpa menyentuh DriveApp
    if (payload.action === "GET_MASUK") {
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        data: getSheetRecords(ss, "SuratMasuk")
      })).setMimeType(ContentService.MimeType.JSON);
    }

    if (payload.action === "GET_KELUAR") {
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        data: getSheetRecords(ss, "SuratKeluar")
      })).setMimeType(ContentService.MimeType.JSON);
    }

    if (payload.action === "GET_DISPOSISI") {
      return ContentService.createTextOutput(JSON.stringify({
        status: "success",
        data: getSheetRecords(ss, "Disposisi")
      })).setMimeType(ContentService.MimeType.JSON);
    }

    const { folderMasuk, folderKeluar } = setupEnvironment();
    let fileUrl = payload.fileUrl || "";

    // Upload File Base64 ke Google Drive sesuai jenis folder
    if (payload.fileBase64 && payload.fileName) {
      const base64Data = payload.fileBase64.split(",")[1];
      const blob = Utilities.newBlob(Utilities.base64Decode(base64Data), payload.mimeType || "application/pdf", payload.fileName);
      const targetFolder = payload.action === "SAVE_MASUK" ? folderMasuk : folderKeluar;
      const file = targetFolder.createFile(blob);
      file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      fileUrl = file.getUrl();
    }

    if (payload.action === "SAVE_MASUK") {
      const sheet = ss.getSheetByName("SuratMasuk");
      sheet.appendRow([payload.id, payload.tglDiterima, payload.asal, payload.nomor, payload.indeks, payload.perihal, payload.tglSurat, payload.jenis, payload.ditujukan, payload.keterangan, payload.fileName, fileUrl]);
    } else if (payload.action === "SAVE_KELUAR") {
      const sheet = ss.getSheetByName("SuratKeluar");
      sheet.appendRow([payload.id, payload.tglPembukuan, payload.nomor, payload.indeks, payload.perihal, payload.tglSurat, payload.jenis, payload.ditujukan, payload.keterangan, payload.fileName, fileUrl]);
    } else if (payload.action === "SAVE_DISPOSISI") {
      const sheet = ss.getSheetByName("Disposisi");
      sheet.appendRow([payload.noDisposisi, payload.noAgenda, payload.nomorSurat, payload.perihal, payload.pengirim, payload.ditujukan, payload.instruksi, payload.tindakLanjut, payload.tglDisposisi, payload.kepalaSekolah, payload.nipKepalaSekolah, payload.pangkatGolongan]);
    }

    return ContentService.createTextOutput(JSON.stringify({ status: "success", fileUrl: fileUrl })).setMimeType(ContentService.MimeType.JSON);
  } catch (err) {
    return ContentService.createTextOutput(JSON.stringify({ status: "error", message: err.toString() })).setMimeType(ContentService.MimeType.JSON);
  }
}
